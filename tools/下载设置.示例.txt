# 下载设置（示例）。要用就把它复制成同目录下的 下载设置.txt，再改。
#
# 这个文件是可选的：没有 下载设置.txt 时，程序全部按下面这些内置默认值跑。
# 键名可以写英文，也可以写中文（parallel = 并行下载数，以此类推）。
# 每行写成 键=值；# 开头是注释；某一行写坏了只跳过那一行，不影响其它设置。
# 文件用哪种编码存都能读（UTF-8 / GBK / 记事本"另存为 Unicode"都认）。

# 同时下载几个视频。1~3 比较稳；越大越容易被 B 站限流。
parallel=2

# 每个视频之间隔几秒再开下一个（可以写小数）。
interval=1

# 连续失败多少个算"被限流"（只统计风控/限速类失败）。
fail_threshold=10

# 判定被限流后先暂停多少秒；之后再撞一次翻倍，最多 30 分钟。
pause_seconds=300

# 每隔几天至少完整校验一次投稿列表（平时走增量同步，只花 1 次请求）。
# 0 = 每次都完整校验（最保守）；1 = 每天完整校验一次。
full_days=7

# 单个视频最多下多少分钟，超时就结束它、下一轮重试。0 = 不限时。
download_timeout_minutes=30

# 这一轮先处理谁：
#   collection_first = 合集/系列优先，再按名字跑 UP主   <-- 默认
#   smallest         = 缺口小的先跑（小号一个个清掉，名单快速变短）
#   largest          = 缺口大的先跑（想尽快啃大号）
#   name             = 按文件夹名，合集排在最后
order=collection_first

# 视频和状态文件放哪（相对项目根目录，也可以写绝对路径；不写这一行就是 videos）。
# 写成盘符根目录(C:\)、系统目录或用户目录本身会被忽略并退回默认值。
# data_root=D:\videos
